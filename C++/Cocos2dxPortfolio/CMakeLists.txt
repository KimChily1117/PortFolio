cmake_minimum_required(VERSION 3.16)
project(AsterIdle LANGUAGES C CXX)

option(ASTER_CORE_ONLY "Build simulation and tests without the renderer" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_library(aster_core Classes/core/GameModel.cpp Classes/core/SkillTimeline.cpp)
target_include_directories(aster_core PUBLIC Classes)
target_compile_features(aster_core PUBLIC cxx_std_17)
if(NOT ANDROID)
    enable_testing()
    add_executable(aster_tests tests/CoreTests.cpp)
    target_link_libraries(aster_tests PRIVATE aster_core)
    add_test(NAME game_rules COMMAND aster_tests)
    add_executable(aster_skill_timeline_tests tests/core/SkillTimelineTests.cpp)
    target_link_libraries(aster_skill_timeline_tests PRIVATE aster_core)
    add_test(NAME skill_timeline COMMAND aster_skill_timeline_tests)
endif()
if(ASTER_CORE_ONLY)
    return()
endif()

set(COCOS2DX_ROOT_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/cocos2d-x" CACHE PATH "Cocos2d-x 4.0")
if(NOT EXISTS "${COCOS2DX_ROOT_PATH}/cocos/CMakeLists.txt")
    message(FATAL_ERROR "Run python tools/bootstrap.py first")
endif()
if(NOT ANDROID)
    add_executable(aster_save_tests tests/SaveTests.cpp Classes/SaveStore.cpp)
    target_include_directories(aster_save_tests PRIVATE "${COCOS2DX_ROOT_PATH}/external")
    target_link_libraries(aster_save_tests PRIVATE aster_core)
    add_test(NAME save_roundtrip COMMAND aster_save_tests)
endif()
set(CMAKE_MODULE_PATH "${COCOS2DX_ROOT_PATH}/cmake/Modules")
set(BUILD_EDITOR_COCOSTUDIO OFF CACHE BOOL "" FORCE)
set(BUILD_EDITOR_SPINE OFF CACHE BOOL "" FORCE)
set(BUILD_EXTENSIONS OFF CACHE BOOL "" FORCE)
set(BUILD_EXT_WEBP OFF CACHE BOOL "Use current WebP decoder below" FORCE)
set(ASTER_WEBP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.deps/libwebp-1.6.0")
if(NOT EXISTS "${ASTER_WEBP_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Run python tools/bootstrap-webp.py first")
endif()
foreach(WEBP_TOOL ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${WEBP_TOOL} OFF CACHE BOOL "" FORCE)
endforeach()
if(ANDROID)
    include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/AndroidWebp.cmake")
else()
    add_subdirectory("${ASTER_WEBP_ROOT}" "${CMAKE_BINARY_DIR}/webp" EXCLUDE_FROM_ALL)
endif()
add_library(aster_cutscene_decoder Classes/WebpAnimation.cpp)
target_include_directories(aster_cutscene_decoder PUBLIC Classes)
target_link_libraries(aster_cutscene_decoder PUBLIC webpdemux)
set_target_properties(aster_cutscene_decoder PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID)
    add_executable(aster_cutscene_tests tests/CutsceneTests.cpp)
    target_link_libraries(aster_cutscene_tests PRIVATE aster_cutscene_decoder)
    set(ASTER_TEST_CUTSCENE "${CMAKE_CURRENT_SOURCE_DIR}/Resources/reference/epicseven/cut/fhd/sk_c1002_3.webp")
    if(EXISTS "${ASTER_TEST_CUTSCENE}")
        add_test(NAME animated_cutscene COMMAND aster_cutscene_tests "${ASTER_TEST_CUTSCENE}")
    endif()
endif()
include(CocosBuildSet)
add_subdirectory("${COCOS2DX_ROOT_PATH}/cocos" "${CMAKE_BINARY_DIR}/engine/cocos")
target_link_libraries(external webp)
target_include_directories(cocos2d PRIVATE "${ASTER_WEBP_ROOT}/src/webp")
file(GLOB ASTER_SPINE_C_SOURCES CONFIGURE_DEPENDS "${COCOS2DX_ROOT_PATH}/cocos/editor-support/spine/*.c")
add_library(aster_spine_runtime STATIC ${ASTER_SPINE_C_SOURCES})
target_include_directories(aster_spine_runtime PUBLIC "${COCOS2DX_ROOT_PATH}/cocos/editor-support")
set_target_properties(aster_spine_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(aster_legacy_pose Classes/LegacySpinePose.cpp)
target_include_directories(aster_legacy_pose PUBLIC Classes PRIVATE "${COCOS2DX_ROOT_PATH}/external")
target_link_libraries(aster_legacy_pose PUBLIC aster_spine_runtime)
set_target_properties(aster_legacy_pose PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID)
    add_executable(aster_legacy_pose_tests tests/LegacyPoseTests.cpp)
    target_link_libraries(aster_legacy_pose_tests PRIVATE aster_legacy_pose)
    add_test(NAME legacy_pose COMMAND aster_legacy_pose_tests)
    add_executable(aster_spine_tests tests/SpineTests.cpp)
    target_link_libraries(aster_spine_tests PRIVATE aster_legacy_pose)
    foreach(ASTER_MODEL portrait/c1001 model/ras)
        get_filename_component(ASTER_MODEL_NAME "${ASTER_MODEL}" NAME)
        set(ASTER_MODEL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/Resources/reference/epicseven/spine/${ASTER_MODEL}")
        if(EXISTS "${ASTER_MODEL_DIR}/${ASTER_MODEL_NAME}.cocos.json")
            add_test(NAME "spine_${ASTER_MODEL_NAME}" COMMAND aster_spine_tests
                "${ASTER_MODEL_DIR}/${ASTER_MODEL_NAME}.cocos.json" "${ASTER_MODEL_DIR}/${ASTER_MODEL_NAME}.atlas")
        endif()
    endforeach()
endif()
set(ASTER_SOURCES Classes/AppDelegate.cpp Classes/GameScene.cpp Classes/FrontendScene.cpp Classes/SaveStore.cpp Classes/ResourceGallery.cpp Classes/SpinePreview.cpp Classes/AnimatedCutscene.cpp Classes/SkillPresentation.cpp Classes/BattleHud.cpp Classes/BattleStageBackground.cpp Classes/ReferenceAudio.cpp Classes/AudioLibraryScene.cpp)
if(ANDROID)
    add_library(AsterIdle SHARED ${ASTER_SOURCES} proj.android/app/jni/main.cpp)
    add_subdirectory("${COCOS2DX_ROOT_PATH}/cocos/platform/android" "${CMAKE_BINARY_DIR}/engine/android")
    target_link_libraries(AsterIdle PRIVATE -Wl,--whole-archive cpp_android_spec -Wl,--no-whole-archive)
else()
    add_executable(AsterIdle WIN32 ${ASTER_SOURCES} proj.win32/main.cpp)
endif()
target_include_directories(AsterIdle PRIVATE Classes)
target_link_libraries(AsterIdle PRIVATE aster_core aster_legacy_pose aster_cutscene_decoder cocos2d)
target_compile_features(AsterIdle PRIVATE cxx_std_17)
setup_cocos_app_config(AsterIdle)
if(WINDOWS)
    cocos_copy_target_dll(AsterIdle)
    set_property(TARGET AsterIdle PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:AsterIdle>")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_custom_target(aster_resources ALL COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/sync-resources.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/Resources" "$<TARGET_FILE_DIR:AsterIdle>/Resources")
    add_dependencies(aster_resources AsterIdle)
endif()
